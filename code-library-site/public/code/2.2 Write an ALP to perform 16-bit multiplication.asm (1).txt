 .MODEL SMALL
.STACK 100H
.DATA
    V1  DW 0000H
    V2  DW 0000H
    R1L DW 0000H
    R1H DW 0000H

    MSG1 DB 0DH, 0AH, 'Enter 16-bit VAR1 (4 hex digits): $'
    MSG2 DB 0DH, 0AH, 'Enter 16-bit VAR2 (4 hex digits): $'
    MSG3 DB 0DH, 0AH, '16-bit Product = $'
.CODE
    MOV AX, @DATA
    MOV DS, AX

    LEA DX, MSG1
    MOV AH, 09H
    INT 21H
    CALL READ16
    MOV V1, AX

    LEA DX, MSG2
    MOV AH, 09H
    INT 21H
    CALL READ16
    MOV V2, AX

    MOV AX, V1
    MUL V2
    MOV R1L, AX
    MOV R1H, DX

    LEA DX, MSG3
    MOV AH, 09H
    INT 21H
    MOV AX, R1H
    CALL DISP16
    MOV AX, R1L
    CALL DISP16

    MOV AH, 4CH
    INT 21H

READ16 PROC
    PUSH BX
    PUSH CX
    MOV CX, 4
    XOR BX, BX
R16L:
    MOV AH, 01H
    INT 21H
    CMP AL, 39H
    JBE R16D
    SUB AL, 07H
R16D:
    SUB AL, 30H
    SHL BX, 4
    OR  BL, AL
    LOOP R16L
    MOV AX, BX
    POP CX
    POP BX
    RET
READ16 ENDP

DISP16 PROC
    PUSH AX
    PUSH BX
    PUSH CX
    MOV BX, AX
    MOV CX, 4
D16L:
    ROL BX, 4
    MOV AL, BL
    AND AL, 0FH
    CALL HEXCH
    LOOP D16L
    POP CX
    POP BX
    POP AX
    RET
DISP16 ENDP

HEXCH PROC
    PUSH AX
    PUSH DX
    AND AL, 0FH
    CMP AL, 9
    JBE HCHD
    ADD AL, 07H
HCHD:
    ADD AL, 30H
    MOV DL, AL
    MOV AH, 02H
    INT 21H
    POP DX
    POP AX
    RET
HEXCH ENDP
END
