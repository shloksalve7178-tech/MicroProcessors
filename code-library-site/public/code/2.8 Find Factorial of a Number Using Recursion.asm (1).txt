
.MODEL SMALL
.STACK 100H
.DATA
    N   DB 00H
    R1  DW 0000H

    MSG1 DB 0DH, 0AH, 'Enter N (2 hex digits): $'
    MSG2 DB 0DH, 0AH, 'Factorial = $'
.CODE
    MOV AX, @DATA
    MOV DS, AX

    LEA DX, MSG1
    MOV AH, 09H
    INT 21H
    CALL READ8
    MOV N, AL

    MOV AL, N
    CALL FACT
    MOV R1, AX

    LEA DX, MSG2
    MOV AH, 09H
    INT 21H
    MOV AX, R1
    CALL DISP16

    MOV AH, 4CH
    INT 21H

FACT PROC
    CMP AL, 1
    JBE FACT1
    PUSH AX
    DEC AL
    CALL FACT
    POP BX
    MUL BL
    RET
FACT1:
    MOV AX, 1
    RET
FACT ENDP

READ8 PROC
    PUSH BX
    PUSH CX
    MOV CX, 2
    XOR BX, BX
R8L:
    MOV AH, 01H
    INT 21H
    CMP AL, 39H
    JBE R8D
    SUB AL, 07H
R8D:
    SUB AL, 30H
    SHL BX, 4
    OR  BL, AL
    LOOP R8L
    MOV AX, BX
    POP CX
    POP BX
    RET
READ8 ENDP

DISP16 PROC
    PUSH AX
    PUSH BX
    PUSH CX
    MOV BX, AX
    MOV CX, 4
D16L:
    ROL BX, 4
    MOV AL, BL
    AND AL, 0FH
    CALL HEXCH
    LOOP D16L
    POP CX
    POP BX
    POP AX
    RET
DISP16 ENDP

HEXCH PROC
    PUSH AX
    PUSH DX
    AND AL, 0FH
    CMP AL, 9
    JBE HCHD
    ADD AL, 07H
HCHD:
    ADD AL, 30H
    MOV DL, AL
    MOV AH, 02H
    INT 21H
    POP DX
    POP AX
    RET
HEXCH ENDP
END
