  .MODEL SMALL
.STACK 100H
.DATA
    N   DB 00H
    A   DB 00H
    B   DB 01H
    C   DB 00H

    MSG1 DB 0DH, 0AH, 'Enter N (2 hex digits): $'
    MSG2 DB 0DH, 0AH, 'Fibonacci Series: $'
    SPACE DB ' $'
.CODE
    MOV AX, @DATA
    MOV DS, AX

    LEA DX, MSG1
    MOV AH, 09H
    INT 21H
    CALL READ8
    MOV N, AL

    LEA DX, MSG2
    MOV AH, 09H
    INT 21H

    MOV AL, A
    CALL DISP8
    LEA DX, SPACE
    MOV AH, 09H
    INT 21H

    MOV AL, B
    CALL DISP8
    LEA DX, SPACE
    MOV AH, 09H
    INT 21H

    MOV CL, N
    SUB CL, 2
FIB:
    MOV AL, A
    ADD AL, B
    MOV C, AL
    MOV AL, B
    MOV A, AL
    MOV AL, C
    MOV B, AL
    MOV AL, C
    CALL DISP8
    LEA DX, SPACE
    MOV AH, 09H
    INT 21H
    DEC CL
    JNZ FIB

    MOV AH, 4CH
    INT 21H

READ8 PROC
    PUSH BX
    PUSH CX
    MOV CX, 2
    XOR BX, BX
R8L:
    MOV AH, 01H
    INT 21H
    CMP AL, 39H
    JBE R8D
    SUB AL, 07H
R8D:
    SUB AL, 30H
    SHL BX, 4
    OR  BL, AL
    LOOP R8L
    MOV AX, BX
    POP CX
    POP BX
    RET
READ8 ENDP

DISP8 PROC
    PUSH AX
    PUSH CX
    MOV CH, AL
    MOV CL, 4
    SHR AL, CL
    CALL HEXCH
    MOV AL, CH
    AND AL, 0FH
    CALL HEXCH
    POP CX
    POP AX
    RET
DISP8 ENDP

HEXCH PROC
    PUSH AX
    PUSH DX
    AND AL, 0FH
    CMP AL, 9
    JBE HCHD
    ADD AL, 07H
HCHD:
    ADD AL, 30H
    MOV DL, AL
    MOV AH, 02H
    INT 21H
    POP DX
    POP AX
    RET
HEXCH ENDP
END
