.MODEL SMALL
.STACK 100H
.DATA
    V1  DB 00H
    V2  DW 0000H
    R1  DB 00H
    R2  DW 0000H

    MSG1 DB 0DH, 0AH, 'Enter 8-bit number (2 hex): $'
    MSG2 DB 0DH, 0AH, 'Enter 16-bit number (4 hex): $'
    MSG3 DB 0DH, 0AH, '8-bit One''s Complement  = $'
    MSG4 DB 0DH, 0AH, '16-bit One''s Complement = $'
.CODE
    MOV AX, @DATA
    MOV DS, AX

    LEA DX, MSG1
    MOV AH, 09H
    INT 21H
    CALL READ8
    MOV V1, AL

    LEA DX, MSG2
    MOV AH, 09H
    INT 21H
    CALL READ16
    MOV V2, AX

    MOV AL, V1
    NOT AL
    MOV R1, AL

    MOV AX, V2
    NOT AX
    MOV R2, AX

    LEA DX, MSG3
    MOV AH, 09H
    INT 21H
    MOV AL, R1
    CALL DISP8

    LEA DX, MSG4
    MOV AH, 09H
    INT 21H
    MOV AX, R2
    CALL DISP16

    MOV AH, 4CH
    INT 21H

READ8 PROC
    PUSH BX
    PUSH CX
    MOV CX, 2
    XOR BX, BX
R8L:
    MOV AH, 01H
    INT 21H
    CMP AL, 39H
    JBE R8D
    SUB AL, 07H
R8D:
    SUB AL, 30H
    SHL BX, 4
    OR  BL, AL
    LOOP R8L
    MOV AX, BX
    POP CX
    POP BX
    RET
READ8 ENDP

READ16 PROC
    PUSH BX
    PUSH CX
    MOV CX, 4
    XOR BX, BX
R16L:
    MOV AH, 01H
    INT 21H
    CMP AL, 39H
    JBE R16D
    SUB AL, 07H
R16D:
    SUB AL, 30H
    SHL BX, 4
    OR  BL, AL
    LOOP R16L
    MOV AX, BX
    POP CX
    POP BX
    RET
READ16 ENDP

DISP8 PROC
    PUSH AX
    PUSH CX
    MOV CH, AL
    MOV CL, 4
    SHR AL, CL
    CALL HEXCH
    MOV AL, CH
    AND AL, 0FH
    CALL HEXCH
    POP CX
    POP AX
    RET
DISP8 ENDP

DISP16 PROC
    PUSH AX
    PUSH BX
    PUSH CX
    MOV BX, AX
    MOV CX, 4
D16L:
    ROL BX, 4
    MOV AL, BL
    AND AL, 0FH
    CALL HEXCH
    LOOP D16L
    POP CX
    POP BX
    POP AX
    RET
DISP16 ENDP

HEXCH PROC
    PUSH AX
    PUSH DX
    AND AL, 0FH
    CMP AL, 9
    JBE HCHD
    ADD AL, 07H
HCHD:
    ADD AL, 30H
    MOV DL, AL
    MOV AH, 02H
    INT 21H
    POP DX
    POP AX
    RET
HEXCH ENDP
END
